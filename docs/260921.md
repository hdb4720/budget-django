# budget/budget/manager_detail_row.py

from django.http import Http404
from django.shortcuts import render


def get_detail_row(respond_to, manager_modal, manager):
    manager_period_id = int(manager_modal["manager_period_id"])
    manager_category_id = int(manager_modal["manager_category_id"])
    
    manager_row = None
    manager_details = []
    manager_selection = None
    for row in manager:
        if (
            row.period_id == manager_period_id and 
            row.category_id == manager_category_id
        ):
            manager_row = row
            manager_details = row.details
            break

    if respond_to == "selection":
        manager_budget_id = int(manager_modal["manager_budget_id"])
        for row in manager_details:
            if row.budget_id == manager_budget_id:
                manager_selection = row

    return {
        "manager_row": manager_row,
        "manager_detail": manager_details,
        "manager_selection": manager_selection,
    }
